function startAttacking(mousePos) {

    // Don't start a new attack interval if one is already running 
    if (player.attackInterval) return; 

    player.attackInterval = setInterval(() => { 

        player.isAttacking=true;
        if (!player.isReloading && player.ammo > 0 && player.playerShootCooldown <= 0) {
            createBullet(playerBullets, player.x + player.size / 2, player.y + player.size / 2, mousePos.x, mousePos.y);
            player.playerShootCooldown = player.maxShootCooldown;
            player.ammo -= 1;

            // Apply rotation and shake effect
            player.gunRotation = -0.1; // Slight rotation to simulate recoil
            player.gunShakeIntensity = 2; // Small shake intensity
            player.gunShakeDuration = 100; // Short duration for the shake effect
        }

    }, player.maxShootCooldown); // Interval time based on cooldown
    
    

    
}

function createBullet(array, x, y, targetX, targetY) {
   
    // Calculate angle towards the mouse 
    const angle = Math.atan2(targetY - y, targetX - x); 
  

    if (array === playerBullets) { 
    

    
        startShake(10, player.attackDamage-5); // Adjust numbers based on desired effect
        if (player.isReloading || (array === playerBullets && player.ammo <= 0)) return;

        createParticles(x, y , angle);
        let gunHoleX = 0;

        if(player.lookingRight){
            gunHoleX = player.currentWeapon.gunHolePositionX
            player.x -= player.currentWeapon.playerKnockBack
        }else{
            gunHoleX = player.currentWeapon.gunHolePositionX - 90
            player.x += player.currentWeapon.playerKnockBack
        }



        //shotGun use
        if(player.currentWeapon.ammoShotNum==1){
            array.push({ 
                x: x+gunHoleX,
                y: y+player.currentWeapon.gunHolePositionY,
                  velocityX: Math.cos(angle) * player.currentWeapon.bulletSpeed, 
                  velocityY: Math.sin(angle) * player.currentWeapon.bulletSpeed,
                  size: player.currentWeapon.bulletSize,
            });
        }else{
            for (let index = 1; index < player.currentWeapon.ammoShotNum+1; index++) {

                let indexChanger = 1;
                if(index%2){
                    indexChanger = -1
                }

                let shotSpread = player.currentWeapon.shotgunSpreadRange

                array.push({ 
                    x: x+gunHoleX,
                    y: y+player.currentWeapon.gunHolePositionY,
                      velocityX: Math.cos(angle) * player.currentWeapon.bulletSpeed + (index*shotSpread)*indexChanger,
                      velocityY: Math.sin(angle) * player.currentWeapon.bulletSpeed + (index*shotSpread)*indexChanger,
                      size: bulletSize,
                });
            }
        }




      } else if (array === enemyBullets) { 
        array.push({
            x: x,
            y: y,
          velocityX: Math.cos(angle) * enemyBulletSpeed, 
          velocityY: Math.sin(angle) * enemyBulletSpeed,
          size: bulletSize,
        });
      }
    if (array === playerBullets) player.ammo -= 1;


    
}



function createParticles(x, y, angle) {

    let partX = 0;
    let partY = 0;

    for (let i = 0; i < 10; i++) { // Create 10 particles per bullet
        
        const anglea = Math.random()*100//* 1; // Random direction
        const speed = Math.random()*player.currentWeapon.shotSpread; // Random speed

        if(player.lookingRight){
            partX = canvas.width /2 + 50 + player.currentWeapon.particLocatX+9;    
        }else{
            partX = (canvas.width /2 + 50 + player.currentWeapon.particLocatX)-90 ;
        }

        particles.push({
            
            x: partX,
            y: canvas.height /2 + 75 + player.currentWeapon.particLocatY,

            velocityX: Math.cos(angle + anglea) * speed,
            velocityY: Math.sin(angle + anglea) * speed,
            size: Math.random() * 5, // Random size
            lifeSpan: Math.random()*5, // Random lifespan
            color: player.currentWeapon.shotColor
        });
    }
}function updateParticles() {
    for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.velocityX;
        p.y += p.velocityY;
        p.lifeSpan--;

        if (p.lifeSpan <= 0) {
            particles.splice(i, 1);
        }
    }
}

function drawParticles(ctx) {
    particles.forEach(p => {
        ctx.fillStyle = p.color;
        ctx.shadowBlur = 10; // Adjust for desired glow size        
        ctx.shadowColor = p.color; // Use the same color for the glow

        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2, true);
        ctx.fill();
    });

    ctx.shadowBlur = 0; // Reset shadowBlur after drawing particles to not affect other canvas elements


}

function updateAndDrawDustParticles(ctx) {
    dustParticles.forEach(particle => {
        particle.x += particle.velocityX ;
        particle.y += particle.velocityY ;

        // Reset particle position if it goes off-screen
        if (particle.x < 0 || particle.x > gameWorld.width || particle.y < 0 || particle.y > gameWorld.height) {
            particle.x = Math.random() * gameWorld.width;
            particle.y = Math.random() * gameWorld.height;
        }

        // Draw particle
        ctx.fillStyle = `rgba(255, 255, 255, ${particle.opacity})`; // White color with variable opacity
        ctx.beginPath();
        ctx.arc(particle.x, particle.y, particle.size, 0, Math.PI * 2, true);
        ctx.fill();

        
    });
}

function performDodge(currentTime) {
    if (player.dodgeCharges > 0) {
        // Use a dodge charge
        player.dodgeCharges--;
        player.isDodging = true;
        player.lastDodgeTime = currentTime; // Update the time since last dodge

        setTimeout(() => {
            player.isDodging = false; // End of dodge - vulnerable again
            // You can trigger tail particle generation here if desired
        }, 500); // Dodge duration, adjust as needed

        // Start recharge timer if it's not already running and if not at max charges
        if (player.dodgeCharges < player.maxDodgeCharges && !player.dodgeRechargeTimer) {
            startDodgeRecharge();
        }
    }

    
}



function startDodgeRecharge() {
    player.dodgeRechargeTimer = setInterval(() => {
        if (player.dodgeCharges < player.maxDodgeCharges) {
            player.dodgeCharges++;
            //console.log(`Dodge Charge Recharged: ${player.dodgeCharges}/${player.maxDodgeCharges}`);
        }
        // Stop the timer if the charges are full
        if (player.dodgeCharges >= player.maxDodgeCharges) {
            clearInterval(player.dodgeRechargeTimer);
            player.dodgeRechargeTimer = null; // Reset the timer
        }
    }, player.dodgeRechargeTime);
}

function updateEntities(array) {
    array.forEach((entity, index) => {
        entity.x += entity.velocityX;
        entity.y += entity.velocityY;

        // Remove if off-screen
        if (entity.x < 0 || entity.x > gameWorld.width || entity.y < 0 || entity.y > gameWorld.height) {
            array.splice(index, 1);
        }
    });
}

    enemies.forEach((enemy, index) => {

 // Decrement the time until the next attack
 if (enemy.timeUntilNextAttack > 0) {
    enemy.timeUntilNextAttack -= deltaTime;
}

        // Check if the enemy is ready to attack
        if (enemy.timeUntilNextAttack <= 0) {
            // Enemy attacks
            createBullet(enemyBullets, enemy.x + enemy.size / 2, enemy.y + enemy.size / 2, player.x, player.y);
            // Reset the attack cooldown
            enemy.timeUntilNextAttack = enemy.attackCooldown;
        }


        if (enemy.hp <= 0) {
            enemies.splice(index, 1);
            totalKills++;

            spawnEnemy()

            gainXP(20); // Example XP gain

        } else {
            const angle = Math.atan2(player.y - enemy.y, player.x - enemy.x);
            enemy.x += Math.cos(angle) * enemy.speed;
            enemy.y += Math.sin(angle) * enemy.speed;
        }
    });
}

function drawHpBar(x, y, width, height, fraction, colorOfHP) {
    const borderWidth = 2; // Thickness of the border

    // Draw the border
    ctx.fillStyle = 'black';
    ctx.fillRect(x - borderWidth, y - borderWidth, width + borderWidth * 2, height + borderWidth * 2);

    ctx.fillStyle = 'gray';
    ctx.fillRect(x, y, width, height);
    ctx.fillStyle = colorOfHP;
    ctx.fillRect(x, y, width * fraction, height);


}



function handleCollisions() {
    playerBullets.forEach((bullet, bulletIndex) => {
        enemies.forEach((enemy, enemyIndex) => {
            if (bullet.x < enemy.x + enemy.size &&
                bullet.x + bullet.size > enemy.x &&
                bullet.y < enemy.y + enemy.size &&
                bullet.y + bullet.size > enemy.y) {
                playerBullets.splice(bulletIndex, 1);


                enemies[enemyIndex].justHit = true;
                enemies[enemyIndex].takenDamage = player.attackDamage + Math.floor(Math.random()*player.currentWeapon.additionalDamage); // Decrease enemy HP by 10
                enemies[enemyIndex].hp -= enemies[enemyIndex].takenDamage 

            enemies[enemyIndex].hitTime = performance.now();


            }
        });
    });

    enemyBullets.forEach((bullet, bulletIndex) => {
        if (bullet.x < PlayercollisionX + PlayercollisionSize &&
            bullet.x + bullet.size > PlayercollisionX &&
            bullet.y < PlayercollisionY + PlayercollisionSize &&
            bullet.y + bullet.size > PlayercollisionY) {
            enemyBullets.splice(bulletIndex, 1);

            if(!player.isDodging){
            player.hp -= 10; // Decrease player HP by 10
            }

            if (player.hp <= 0) {
                alert("Game Over!");
                document.location.reload();
            }
        }
 
    });
}

}

function gameLoop(timestamp) {
    updateGameTime(); // Update the game time each frame

    
    if (!timestamp) {  // Check if lastTime is undefined (first run)
        timestamp = 0; 
    }
    

    const deltaTime = timestamp - lastTime;
    lastTime = timestamp;


    animationTimer += deltaTime * 3;
    if (animationTimer >= frameInterval) {
        animationTimer = 0; // Subtract excess for smooth animation
        currentFrame++; // Increment the frame
        PlayercurrentFrame++;


        if (currentFrame >= numberOfFrames) {
            currentFrame = 0; // Reset for looping
        }

        if(PlayercurrentFrame >= playerNumberOfframes){
            PlayercurrentFrame=0;
        }
        


    }
    

    update(deltaTime);    
    updateCollisionBox();

    draw(currentFrame, deltaTime);
    updateParticles();
    drawParticles(ctx);












    requestAnimationFrame(gameLoop);

}
